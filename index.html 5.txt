<script>

let team1 = {
    name:"",
    players:[],
    captain:""
};

let team2 = {
    name:"",
    players:[],
    captain:""
};

let match = null;

let playerStats = {};

let striker = null;
let nonStriker = null;
let bowler = null;


/* =========================
   TEAM FUNCTIONS
========================= */

function addPlayer(teamNo){

    let input =
        document.getElementById(
            teamNo === 1
            ? "team1PlayerInput"
            : "team2PlayerInput"
        );

    let name = input.value.trim();

    if(!name){
        alert("Please enter player name.");
        return;
    }

    let team = teamNo === 1 ? team1 : team2;

    if(team.players.includes(name)){
        alert("This player already exists.");
        return;
    }

    team.players.push(name);

    input.value="";

    updateTeamDisplay(teamNo);
}


function removePlayer(teamNo,index){

    let team = teamNo === 1 ? team1 : team2;

    team.players.splice(index,1);

    if(team.captain === undefined){
        team.captain="";
    }

    updateTeamDisplay(teamNo);
}


function updateTeamDisplay(teamNo){

    let team = teamNo === 1 ? team1 : team2;

    let box =
        document.getElementById(
            teamNo === 1
            ? "team1Players"
            : "team2Players"
        );

    box.innerHTML="";

    team.players.forEach((player,index)=>{

        let div=document.createElement("div");

        div.className="playerRow";

        div.innerHTML=`
            <span>${escapeHTML(player)}</span>
            <button class="red"
            onclick="removePlayer(${teamNo},${index})">
            ×
            </button>
        `;

        box.appendChild(div);
    });

    let captainSelect =
        document.getElementById(
            teamNo === 1
            ? "team1Captain"
            : "team2Captain"
        );

    let oldCaptain = team.captain;

    captainSelect.innerHTML =
        `<option value="">Select Captain</option>`;

    team.players.forEach(player=>{

        let option=document.createElement("option");

        option.value=player;
        option.textContent=player;

        captainSelect.appendChild(option);
    });

    captainSelect.value=oldCaptain;

    captainSelect.onchange=function(){

        team.captain=this.value;

    };
}


/* =========================
   TOSS
========================= */

function startToss(){

    team1.name =
        document.getElementById("team1Name").value.trim();

    team2.name =
        document.getElementById("team2Name").value.trim();

    if(!team1.name || !team2.name){
        alert("Please enter both team names.");
        return;
    }

    if(team1.players.length < 2 ||
       team2.players.length < 2){

        alert("Please add at least 2 players in each team.");
        return;
    }

    team1.captain =
        document.getElementById("team1Captain").value;

    team2.captain =
        document.getElementById("team2Captain").value;

    if(!team1.captain || !team2.captain){

        alert("Please choose captain for both teams.");
        return;
    }

    let toss =
        document.getElementById("tossWinner");

    toss.innerHTML=`
        <option value="">Select Toss Winner</option>
        <option value="1">${escapeHTML(team1.name)}</option>
        <option value="2">${escapeHTML(team2.name)}</option>
    `;

    document.getElementById("setupCard")
        .classList.add("hidden");

    document.getElementById("tossCard")
        .classList.remove("hidden");
}


/* =========================
   START MATCH
========================= */

function startMatch(){

    let winner =
        document.getElementById("tossWinner").value;

    let decision =
        document.getElementById("tossDecision").value;

    if(!winner || !decision){

        alert("Please complete the toss.");
        return;
    }

    let tossTeam =
        winner === "1" ? 1 : 2;

    let firstBatting;

    if(decision === "bat"){

        firstBatting=tossTeam;

    }else{

        firstBatting =
            tossTeam === 1 ? 2 : 1;
    }

    let overs =
        parseInt(
            document.getElementById("oversSelect").value
        );

    match={
        innings:1,
        battingTeam:firstBatting,
        firstBattingTeam:firstBatting,
        runs:0,
        wickets:0,
        balls:0,
        overs:overs,
        finished:false,
        innings1:null,
        innings2:null,
        history:[]
    };

    initializeStats();

    document.getElementById("tossCard")
        .classList.add("hidden");

    document.getElementById("matchArea")
        .classList.remove("hidden");

    setupInningsPlayers();

    updateUI();

    alert(
        "Toss completed!\n\n" +
        getTeam(firstBatting).name +
        " will bat first."
    );
}


/* =========================
   STATS
========================= */

function initializeStats(){

    playerStats={};

    team1.players.forEach(player=>{

        playerStats["1_"+player]={
            team:1,
            name:player,
            runs:0,
            balls:0,
            fours:0,
            sixes:0,
            out:false
        };

    });

    team2.players.forEach(player=>{

        playerStats["2_"+player]={
            team:2,
            name:player,
            runs:0,
            balls:0,
            fours:0,
            sixes:0,
            out:false
        };

    });
}


function statKey(team,player){

    return team+"_"+player;
}


function getStat(team,player){

    return playerStats[statKey(team,player)];
}


/* =========================
   TEAM
========================= */

function getTeam(teamNo){

    return teamNo === 1 ? team1 : team2;
}


/* =========================
   INNINGS PLAYERS
========================= */

function setupInningsPlayers(){

    striker=null;
    nonStriker=null;
    bowler=null;

    let batting=getTeam(match.battingTeam);
    let bowling=getTeam(
        match.battingTeam === 1 ? 2 : 1
    );

    fillBattingSelects(batting);
    fillBowlerSelect(bowling);

    document.getElementById("strikerSelect").value="";
    document.getElementById("nonStrikerSelect").value="";
    document.getElementById("bowlerSelect").value="";
}


function fillBattingSelects(team){

    let s=document.getElementById("strikerSelect");
    let ns=document.getElementById("nonStrikerSelect");

    s.innerHTML=`<option value="">Select Striker</option>`;
    ns.innerHTML=`<option value="">Select Non-Striker</option>`;

    team.players.forEach(player=>{

        let st=getStat(match.battingTeam,player);

        if(st && !st.out){

            s.innerHTML +=
                `<option value="${escapeAttribute(player)}">
                ${escapeHTML(player)}
                </option>`;

            ns.innerHTML +=
                `<option value="${escapeAttribute(player)}">
                ${escapeHTML(player)}
                </option>`;
        }

    });
}


function fillBowlerSelect(team){

    let select=document.getElementById("bowlerSelect");

    select.innerHTML=
        `<option value="">Select Bowler</option>`;

    team.players.forEach(player=>{

        select.innerHTML +=
            `<option value="${escapeAttribute(player)}">
            ${escapeHTML(player)}
            </option>`;

    });
}


function setPlayers(){

    let s =
        document.getElementById("strikerSelect").value;

    let ns =
        document.getElementById("nonStrikerSelect").value;

    let b =
        document.getElementById("bowlerSelect").value;

    if(!s || !ns || !b){

        alert(
            "Please select striker, non-striker and bowler."
        );

        return;
    }

    if(s===ns){

        alert(
            "Striker and non-striker must be different."
        );

        return;
    }

    striker=s;
    nonStriker=ns;
    bowler=b;

    updateUI();

    alert("Players selected successfully!");
}


/* =========================
   STRIKE
========================= */

function swapStrike(){

    let temp=striker;

    striker=nonStriker;
    nonStriker=temp;
}


/* =========================
   BOOM BOOM
========================= */

function isBoomBall(){

    return ((match.balls + 1) % 6 === 0);
}


/* =========================
   INNINGS FINISHED
========================= */

function inningsFinished(){

    if(match.balls >= match.overs * 6){
        return true;
    }

    let batting=getTeam(match.battingTeam);

    if(match.wickets >= batting.players.length - 1){
        return true;
    }

    return false;
}


/* =========================
   SCORE RUN
========================= */

function scoreRun(runs){

    if(!match || match.finished){
        return;
    }

    if(!striker || !nonStriker || !bowler){

        alert(
            "Please select striker, non-striker and bowler first."
        );

        return;
    }

    if(inningsFinished()){

        finishCurrentInnings();
        return;
    }

    let boom=isBoomBall();

    let actualRuns =
        boom ? runs * 2 : runs;

    match.runs += actualRuns;

    let stat=getStat(
        match.battingTeam,
        striker
    );

    if(stat){

        stat.runs += actualRuns;
        stat.balls++;

        if(runs===4){
            stat.fours++;
        }

        if(runs===6){
            stat.sixes++;
        }
    }

    match.balls++;

    match.history.push({

        type:"run",
        raw:runs,
        actual:actualRuns,
        boom:boom,
        ball:match.balls,
        striker:striker,
        bowler:bowler

    });

    /*
      IMPORTANT:
      Strike is calculated from ORIGINAL runs,
      not doubled Boom Boom runs.
    */

    if(runs % 2 === 1){

        swapStrike();

    }

    /*
      At end of every over:
      swap strike one more time.
    */

    if(match.balls % 6 === 0){

        swapStrike();

        setTimeout(()=>{

            alert(
                "🏏 OVER COMPLETED!\n\n" +
                "Strike changed automatically."
            );

        },50);
    }

    updateUI();

    if(checkChase()) return;

    if(inningsFinished()){

        setTimeout(
            finishCurrentInnings,
            100
        );
    }
}


/* =========================
   EXTRAS
========================= */

function extraRun(runs,type){

    if(!match || match.finished){
        return;
    }

    if(!striker || !nonStriker || !bowler){

        alert(
            "Please select striker, non-striker and bowler first."
        );

        return;
    }

    match.runs += runs;

    match.history.push({

        type:"extra",
        extra:type,
        runs:runs,
        ball:match.balls

    });

    updateUI();
}


/* =========================
   WICKET
========================= */

function wicket(){

    if(!match || match.finished){
        return;
    }

    if(!striker || !nonStriker || !bowler){

        alert(
            "Please select striker, non-striker and bowler first."
        );

        return;
    }

    let boom=isBoomBall();

    if(boom){

        match.runs -= 5;

        if(match.runs < 0){
            match.runs=0;
        }

        alert(
            "💥 BOOM BOOM WICKET!\n\n" +
            "Batting team gets -5 runs."
        );
    }

    let stat=getStat(
        match.battingTeam,
        striker
    );

    if(stat){

        stat.out=true;
        stat.balls++;
    }

    match.wickets++;
    match.balls++;

    match.history.push({

        type:"wicket",
        boom:boom,
        ball:match.balls,
        player:striker,
        bowler:bowler

    });

    /*
      Wicket ball is legal.
      If it is the last ball of over,
      end-over strike change happens once.
    */

    if(match.balls % 6 === 0){

        swapStrike();
    }

    updateUI();

    if(inningsFinished()){

        setTimeout(
            finishCurrentInnings,
            200
        );

        return;
    }

    /*
      New batter selection.
    */

    let batting=getTeam(match.battingTeam);

    let available=batting.players.filter(player=>{

        let st=getStat(match.battingTeam,player);

        return st && !st.out &&
               player !== nonStriker;

    });

    if(available.length===0){

        finishCurrentInnings();
        return;
    }

    let newBatter=prompt(
        "WICKET!\n\nSelect new batter:\n\n" +
        available.join("\n")
    );

    if(!newBatter ||
       !available.includes(newBatter)){

        alert(
            "Please choose a valid new batter from the available players."
        );

        updateUI();
        return;
    }

    striker=newBatter;

    updateUI();
}


/* =========================
   CHASE CHECK
========================= */

function checkChase(){

    if(match.innings !== 2){
        return false;
    }

    let target =
        match.innings1.runs + 1;

    if(match.runs >= target){

        let batting=getTeam(match.battingTeam);

        let wicketsRemaining =
            batting.players.length - 1 - match.wickets;

        showResult(
            batting.name +
            " won by " +
            wicketsRemaining +
            " wicket" +
            (wicketsRemaining===1 ? "" : "s")
        );

        return true;
    }

    return false;
}


/* =========================
   END CURRENT INNINGS
========================= */

function finishCurrentInnings(){

    if(!match || match.finished){
        return;
    }

    if(match.innings===1){

        match.innings1={

            team:match.battingTeam,
            runs:match.runs,
            wickets:match.wickets,
            balls:match.balls,
            history:[...match.history]

        };

        /*
          SECOND INNINGS
        */

        match.innings=2;

        match.battingTeam =
            match.battingTeam === 1 ? 2 : 1;

        match.runs=0;
        match.wickets=0;
        match.balls=0;
        match.history=[];

        striker=null;
        nonStriker=null;
        bowler=null;

        setupInningsPlayers();

        updateUI();

        let target=
            match.innings1.runs + 1;

        alert(
            "🏏 FIRST INNINGS COMPLETED!\n\n" +
            getTeam(match.innings1.team).name +
            " scored " +
            match.innings1.runs +
            "/" +
            match.innings1.wickets +
            "\n\n" +
            getTeam(match.battingTeam).name +
            " needs " +
            target +
            " runs to win.\n\n" +
            "SECOND INNINGS STARTED!"
        );

        return;
    }


    /*
      SECOND INNINGS COMPLETED
    */

    match.innings2={

        team:match.battingTeam,
        runs:match.runs,
        wickets:match.wickets,
        balls:match.balls,
        history:[...match.history]

    };

    determineWinner();
}


/* =========================
   WINNER
========================= */

function determineWinner(){

    let first=match.innings1;
    let second=match.innings2;

    let firstTeam=getTeam(first.team);
    let secondTeam=getTeam(second.team);

    let resultText="";

    if(second.runs > first.runs){

        let wicketsRemaining =
            secondTeam.players.length - 1 - second.wickets;

        resultText =
            secondTeam.name +
            " won by " +
            wicketsRemaining +
            " wicket" +
            (wicketsRemaining===1 ? "" : "s");

    }
    else if(second.runs < first.runs){

        let runDifference =
            first.runs - second.runs;

        resultText =
            firstTeam.name +
            " won by " +
            runDifference +
            " run" +
            (runDifference===1 ? "" : "s");

    }
    else{

        resultText="Match Tied 🤝";

    }

    showResult(resultText);
}


/* =========================
   SHOW RESULT
========================= */

function showResult(text){

    match.finished=true;

    let result=document.getElementById("resultArea");

    result.className="result";

    result.innerHTML=`

        <h1>🏆 MATCH RESULT</h1>

        <h2 style="color:white;">
            ${escapeHTML(text)}
        </h2>

        <p>
            ${escapeHTML(team1.name)}
            :
            ${match.innings1 ? match.innings1.runs : match.runs}
            /
            ${match.innings1 ? match.innings1.wickets : match.wickets}
        </p>

        ${
            match.innings2
            ?
            `<p>
                ${escapeHTML(team2.name)}
                :
                ${match.innings2.runs}/${match.innings2.wickets}
            </p>`
            :
            ""
        }

    `;

    result.classList.remove("hidden");

    saveMatch(true);

    updateUI();

    window.scrollTo({
        top:result.offsetTop-20,
        behavior:"smooth"
    });
}


/* =========================
   FINISH MATCH BUTTON
========================= */

function finishMatch(){

    if(!match){
        return;
    }

    if(match.finished){
        alert("Match is already finished.");
        return;
    }

    if(match.innings===1){

        let confirmFinish=
            confirm(
                "First innings is still running.\n\n" +
                "Finish match anyway?"
            );

        if(!confirmFinish){
            return;
        }

        match.innings1={

            team:match.battingTeam,
            runs:match.runs,
            wickets:match.wickets,
            balls:match.balls

        };

        match.innings2={

            team:
                match.battingTeam===1 ? 2 : 1,

            runs:0,
            wickets:0,
            balls:0

        };

        determineWinner();

        return;
    }

    match.innings2={

        team:match.battingTeam,
        runs:match.runs,
        wickets:match.wickets,
        balls:match.balls

    };

    determineWinner();
}

   UI UPDATE
========================= */

function updateUI(){

    if(!match){
        return;
    }

    let team=getTeam(match.battingTeam);

    document.getElementById("battingTeamTitle")
        .textContent=team.name;

    document.getElementById("runs")

    document.getElementById("wickets")
        .textContent=match.wickets;

    document.getElementById("maxOvers")

        Math.floor(match.balls/6);

    let ballNumber=
        match.balls%6;

    document.getElementById("overs")
        .textContent=
        overNumber+"."+ballNumber;

        .textContent=striker || "-";

    document.getElementById("nonStrikerName")
        .textContent=nonStriker || "-";

    document.getElementById("bowlerName")
        .textContent=bowler || "-";

    document.getElementById("inningsStatus")
        .textContent=
        match.innings===1
        ? "🏏 FIRST INNINGS"
        : "🏏 SECOND INNINGS";

        let target=
            match.innings1.runs+1;

            target-match.runs;

            .textContent=
            "🎯 Target: "+
            target+
            " | Need: "+

    }else{

        document.getElementById("targetText")
            .textContent="";
    }

    let boom=isBoomBall();

    document.getElementById("boomBanner")
        .style.display=
        boom ? "block" : "none";

    updateAnalysis();
}

/* =========================
   SAVE MATCH
========================= */

function saveMatch(silent=false){

    if(!match){
        return;
    }

    let savedMatches=
        JSON.parse(
            localStorage.getItem("boomBoomMatches") || "[]"
        );

    let copy={

        id:Date.now(),

        team1:{
            name:team1.name,
            captain:team1.captain,
            players:[...team1.players]
        },

        team2:{
            name:team2.name,
            captain:team2.captain,
            players:[...team2.players]
        },

        match:JSON.parse(JSON.stringify(match)),

        playerStats:
            JSON.parse(JSON.stringify(playerStats)),

        date:new Date().toLocaleString()

    };

    savedMatches.unshift(copy);

    /*
      Keep only latest 30 matches.
    */

    savedMatches=savedMatches.slice(0,30);

    localStorage.setItem(
        "boomBoomMatches",
        JSON.stringify(savedMatches)
    );

    loadHistory();

    if(!silent){

        alert("💾 Match saved successfully!");
    }
}


/* =========================
   HISTORY
========================= */

function loadHistory(){

    let historyBox=
        document.getElementById("history");

    let matches=
        JSON.parse(
            localStorage.getItem("boomBoomMatches") || "[]"
        );

    if(matches.length===0){

        historyBox.innerHTML=
            "<p>No saved matches yet.</p>";

        return;
    }

    historyBox.innerHTML="";

    matches.forEach((item,index)=>{

        let m=item.match;

        let result="Match in progress";

        if(m.innings1 && m.innings2){

            if(m.innings2.runs > m.innings1.runs){

                let wickets=
                    item.team2.players.length-1-
                    m.innings2.wickets;

                result=
                    item.team2.name+
                    " won by "+
                    wickets+
                    " wickets";

            }else if(
                m.innings2.runs <
                m.innings1.runs
            ){

                result=
                    item.team1.name+
                    " won by "+
                    (
                        m.innings1.runs-
                        m.innings2.runs
                    )+
                    " runs";

            }else{

                result="Match Tied";

            }
        }

        let div=
            document.createElement("div");

        div.className="historyItem";

        div.innerHTML=`

            <b>${escapeHTML(item.team1.name)}
            vs
            ${escapeHTML(item.team2.name)}</b>

            <br>

            ${escapeHTML(result)}

            <br>

            <span class="small">
            ${escapeHTML(item.date)}
            </span>

            <br>

            <button class="primary"
            onclick="analyseSaved(${index})">
            📊 Analyse
            </button>

            <button class="red"
            onclick="deleteSaved(${index})">
            🗑 Delete
            </button>

        `;

        historyBox.appendChild(div);
    });
}


function analyseSaved(index){

    let matches=
        JSON.parse(
            localStorage.getItem("boomBoomMatches") || "[]"
        );

    let item=matches[index];

    let text="";

    Object.values(item.playerStats).forEach(st=>{

        text +=
            st.name+
            " - Runs: "+
            st.runs+
            ", Balls: "+
            st.balls+
            ", 4s: "+
            st.fours+
            ", 6s: "+
            st.sixes+
            (st.out ? ", OUT" : ", NOT OUT")+
            "\n";

    });

    alert(
        "📊 PLAYER ANALYSIS\n\n"+
        text
    );
}


function deleteSaved(index){

    if(!confirm("Delete this saved match?")){
        return;
    }

    let matches=
        JSON.parse(
            localStorage.getItem("boomBoomMatches") || "[]"
        );

    matches.splice(index,1);

    localStorage.setItem(
        "boomBoomMatches",
        JSON.stringify(matches)
    );

    loadHistory();
}

/* =========================
   SECURITY HELPERS
========================= */

function escapeHTML(text){

    return String(text)
        .replace(/&/g,"&amp;")
        .replace(/</g,"&lt;")
        .replace(/>/g,"&gt;")
        .replace(/"/g,"&quot;")
        .replace(/'/g,"&#039;");
}


function escapeAttribute(text){

    return String(text)
        .replace(/"/g,"&quot;")
        .replace(/'/g,"&#039;");
}
